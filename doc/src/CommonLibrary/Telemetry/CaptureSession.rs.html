2//! Emit `land:<tier>:session:start` with pid / OS / arch. Called once
3//! by `Initialize::Fn` so the Boot & Startup Performance dashboard
4//! sees one start event per sidecar process.
5
6use crate::Telemetry::{CaptureEvent, Client};
7
8pub fn Fn() {
9	let TierStr = Client::TIER.get().map(|T| T.AsStr()).unwrap_or("common");
10
11	let EventName = format!("land:{}:session:start", TierStr);
12
13	let Pid = format!("{}", std::process::id());
14
15	CaptureEvent::Fn(
16		&EventName,
17		Some(vec![
18			("pid", Pid.as_str()),
19			("os", std::env::consts::OS),
20			("arch", std::env::consts::ARCH),
21		]),
22	);
23}
