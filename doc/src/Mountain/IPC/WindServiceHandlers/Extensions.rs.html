12/// VS Code's `ExtensionType` enum - mirror the numeric values used by the
13/// renderer's `getInstalled(type?)` IPC so the filter in `GetInstalledArgs`
14/// matches what the channel client sends.
15///
16/// `src/vs/platform/extensions/common/extensions.ts` in the pinned VS Code
17/// dependency:
18/// ```ts
19/// export const enum ExtensionType { System = 0, User = 1 }
20/// ```
21const EXTENSION_TYPE_SYSTEM:u8 = 0;
22const EXTENSION_TYPE_USER:u8 = 1;
23
24/// Return scanned extensions reshaped as VS Code's `ILocalExtension[]`
25/// so `ExtensionManagementChannelClient.getInstalled` can destructure
26/// `extension.identifier.id`, `extension.manifest.*`, and
27/// `extension.location` without blowing up.
28///
29/// # Argument contract
30///
31/// `args[0]` is the optional `ExtensionType` filter VS Code passes in:
32/// - `0` (System) → only return built-in extensions.
33/// - `1` (User) → only return VSIX-installed extensions.
34/// - `null` / missing → return every known extension.
35///
36/// Previously this filter was silently dropped and every call returned the
37/// full list hardcoded as `type: 0, isBuiltin: true`. That produced three
38/// cascading symptoms:
39///   1. VSIX-installed extensions (e.g. `Anthropic.claude-code`) showed up
40///      under "Built-in" in the Extensions sidebar and had no Uninstall action
41///      because the UI keys off `type === User`.
42///   2. The trusted-publishers boot migration iterated every extension as User
43///      and attempted `manifest.publisher.toLowerCase()` against System
44///      manifests.
45///   3. `extensions:scanUserExtensions` (which shares the user-only semantic)
46///      returned zero, making the "Install from VSIX…" refresh appear to do
47///      nothing even when the install itself succeeded.
48pub async fn handle_extensions_get_installed(
49	runtime:Arc<ApplicationRunTime>,
50	args:Vec<Value>,
51) -> Result<Value, String> {
52	let TypeFilter:Option<u8> = args.first().and_then(|V| V.as_u64()).map(|N| N as u8);
53
54	let Extensions = runtime
55		.Environment
56		.GetExtensions()
57		.await
58		.map_err(|Error| format!("extensions:getInstalled failed: {}", Error))?;
59
60	let Wrapped:Vec<Value> = Extensions
61		.into_iter()
62		.filter_map(|Manifest| {
63			// `isBuiltin` is authored by the scanner; default to `true` for
64			// safety when the field is missing (matches the pre-filter
65			// hardcoded behaviour so we never drop an extension the renderer
66			// used to see).
67			let IsBuiltin = Manifest.get("isBuiltin").and_then(Value::as_bool).unwrap_or(true);
68			let ExtensionType = if IsBuiltin { EXTENSION_TYPE_SYSTEM } else { EXTENSION_TYPE_USER };
69
70			if let Some(Wanted) = TypeFilter {
71				if Wanted != ExtensionType {
72					return None;
73				}
74			}
75
76			let Publisher = Manifest
77				.get("publisher")
78				.and_then(Value::as_str)
79				.filter(|S| !S.is_empty())
80				.unwrap_or("unknown")
81				.to_string();
82			let Name = Manifest
83				.get("name")
84				.and_then(Value::as_str)
85				.filter(|S| !S.is_empty())
86				.unwrap_or("unknown")
87				.to_string();
88			let Id = format!("{}.{}", Publisher, Name);
89
90			// VS Code's `URI.revive()` is a no-op on strings, so the scanner's
91			// `file://…` raw URL has to be reshaped into `UriComponents` here -
92			// otherwise every `location.fsPath` / `location.scheme` access in
93			// the sidebar silently returns `undefined` and the whole batch is
94			// filtered out. Normalize once, reuse for both the top-level
95			// `location` and the mirror inside `manifest.extensionLocation` so
96			// callers that read either field get the same shape.
97			let Location = NormalizeUri(Manifest.get("extensionLocation"));
98			// Guarantee the manifest is an object with non-empty `publisher`,
99			// `name` and `version` fields before it reaches the renderer. VS
100			// Code runs a trusted-publishers migration at first-boot
101			// (`extensions.contribution.ts`) that unconditionally calls
102			// `extension.manifest.publisher.toLowerCase()`; any missing
103			// `manifest` object, or a manifest with `publisher === undefined`,
104			// crashes the webview with
105			// `TypeError: undefined is not an object (evaluating 'manifest.publisher')`
106			// before the workbench can render a single pixel. A non-object
107			// value here (null / Value::Null from upstream scan failures) is
108			// replaced with a bare skeleton so the renderer always has shape.
109			let mut Manifest = match Manifest {
110				Value::Object(_) => Manifest,
111				_ => json!({}),
112			};
113			if let Value::Object(ref mut Map) = Manifest {
114				Map.insert("extensionLocation".to_string(), Location.clone());
115				Map.entry("publisher".to_string()).or_insert_with(|| json!(Publisher.clone()));
116				Map.entry("name".to_string()).or_insert_with(|| json!(Name.clone()));
117				Map.entry("version".to_string()).or_insert_with(|| json!("0.0.0"));
118			}
119
120			Some(json!({
121				// IExtension (base)
122				"type": ExtensionType,
123				"isBuiltin": IsBuiltin,
124				"identifier": { "id": Id },
125				"manifest": Manifest,
126				"location": Location,
127				"targetPlatform": "undefined",
128				"isValid": true,
129				"validations": [],
130				"preRelease": false,
131				// ILocalExtension (extras)
132				"isWorkspaceScoped": false,
133				"isMachineScoped": false,
134				"isApplicationScoped": false,
135				"publisherId": null,
136				"isPreReleaseVersion": false,
137				"hasPreReleaseVersion": false,
138				"private": false,
139				"updated": false,
140				"pinned": false,
141				"forceAutoUpdate": false,
142				// `source` distinguishes the disk origin: built-ins ship with
143				// the bundle ("system"); VSIX-installed extensions live under
144				// `~/.land/extensions/*` ("vsix"). The sidebar keys off this
145				// for the "Uninstall" gesture.
146				"source": if IsBuiltin { "system" } else { "vsix" },
147				"size": 0,
148			}))
149		})
150		.collect();
151
152	dev_log!(
153		"extensions",
154		"extensions:getInstalled type={:?} returning {} ILocalExtension-shaped entries",
155		TypeFilter,
156		Wrapped.len()
157	);
158
159	Ok(json!(Wrapped))
160}

162/// Return metadata for all scanned extensions.
163pub async fn handle_extensions_get_all(runtime:Arc<ApplicationRunTime>) -> Result<Value, String> {
164	let Extensions = runtime
165		.Environment
166		.GetExtensions()
167		.await
168		.map_err(|Error| format!("extensions:getAll failed: {}", Error))?;
169
170	dev_log!("extensions", "extensions:getAll returning {} extensions", Extensions.len());
171	if let Some(First) = Extensions.first() {
172		dev_log!(
173			"extensions",
174			"extensions:getAll sample: {}",
175			serde_json::to_string(First)
176				.unwrap_or_default()
177				.chars()
178				.take(300)
179				.collect::<String>()
180		);
181	}
182	Ok(json!(Extensions))
183}

185/// Return metadata for a single extension by ID.
186pub async fn handle_extensions_get(runtime:Arc<ApplicationRunTime>, args:Vec<Value>) -> Result<Value, String> {
187	let Id = args
188		.first()
189		.and_then(|V| V.as_str())
190		.ok_or_else(|| "extensions:get requires string id as first argument".to_string())?
191		.to_string();
192
193	let Extension = runtime
194		.Environment
195		.GetExtension(Id)
196		.await
197		.map_err(|Error| format!("extensions:get failed: {}", Error))?;
198
199	Ok(Extension.unwrap_or(Value::Null))
200}
