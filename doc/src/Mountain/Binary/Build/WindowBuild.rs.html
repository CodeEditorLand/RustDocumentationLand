1//! # Window Build Module
2//!
3//! Creates and configures the main application window.
4
5use tauri::{App, WebviewUrl, WebviewWindowBuilder, Wry};
6
7use crate::IPC::WindServiceHandlers::Utilities::RecentlyOpened::ReadRecentlyOpened;

9/// Creates and configures the main application window.
10///
11/// # Arguments
12///
13/// * `Application` - The Tauri application instance
14/// * `LocalhostUrl` - The localhost URL for the webview content
15///
16/// # Returns
17///
18/// A configured `WebviewWindow<Wry>` instance.
19///
20/// # Platform-Specific Behavior
21///
22/// - **macOS**: `TitleBarStyle::Overlay` + `hidden_title(true)` keeps the
23///   traffic-light buttons at the top-left but hides the native title bar
24///   strip, so VS Code's custom titlebar (which has `-webkit-app-region: drag`
25///   baked into its CSS) lights up the entire top row as a drag region. The
26///   previous `maximized(true)` was the direct cause of "can't drag the editor
27///   around" - maximized macOS windows are pinned to the screen and refuse all
28///   drag events, regardless of `app-region` CSS.
29/// - **Windows / Linux**: `decorations(false)` keeps the window chrome-less so
30///   the workbench draws its own. We still start `resizable(true)` so the
31///   window can be moved by the drag region.
32/// - **Debug builds**: DevTools auto-open.
33pub fn WindowBuild(Application:&mut App, LocalhostUrl:String) -> tauri::WebviewWindow<Wry> {
34	// Restore the most-recently-opened folder so the webview boots
35	// directly into the workspace. Without this, every launch lands
36	// on the Welcome tab, the user clicks "Open Folder", and the
37	// `pickFolderAndOpen` handler fires `Window.navigate()` - a hard
38	// reload that wipes workbench state mid-initialisation and is
39	// the direct cause of the purple splash flash, stuttering paint,
40	// empty `@builtin` sidebar, and broken keybindings (every layer
41	// has to boot twice and the second pass often loses references
42	// to the first). Using `?folder=...` in the initial URL skips
43	// that destructive round-trip.
44	let InitialUrl = BuildInitialUrl(&LocalhostUrl);
45	let WindowUrl = WebviewUrl::External(InitialUrl.parse().expect("FATAL: Failed to parse initial webview URL"));
46
47	// Configure window builder with base settings.
48	//
49	// `visible(false)` is the hidden-until-ready pattern. Tauri's default
50	// is to show the window the instant it's built, which paints the native
51	// chrome + Base.astro's `#1e1e1e` inline background + VS Code theme CSS
52	// + workbench DOM in four separate repaints over the first ~200 ms -
53	// observed as the "purple/dark flash" and panel-pop flicker.
54	//
55	// Mountain shows the window explicitly when the frontend's
56	// `lifecycle:advancePhase(3)` (Restored) arrives, which fires after
57	// `.monaco-workbench` is attached and the first frame is ready. A 3 s
58	// safety timer in `AppLifecycle` guarantees the window appears even if
59	// Sky crashes before signalling phase 3.
60	let mut WindowBuilder = WebviewWindowBuilder::new(Application, "main", WindowUrl)
61		.use_https_scheme(false)
62		.initialization_script("")
63		.zoom_hotkeys_enabled(true)
64		.browser_extensions_enabled(false)
65		.title("Mountain")
66		.resizable(true)
67		.inner_size(1400.0, 900.0)
68		.shadow(true)
69		.visible(false);
70
71	#[cfg(target_os = "macos")]
72	{
73		// Overlay style lets VS Code's custom titlebar paint behind the
74		// traffic-light buttons. `hidden_title(true)` suppresses the OS
75		// title text so it doesn't collide with the workbench menubar.
76		// `decorations(true)` is REQUIRED for the traffic lights to
77		// render - turning decorations off also removes the buttons and
78		// breaks the native drag + resize handles entirely on macOS.
79		WindowBuilder = WindowBuilder
80			.title_bar_style(tauri::TitleBarStyle::Overlay)
81			.hidden_title(true)
82			.decorations(true);
83	}
84
85	#[cfg(any(target_os = "windows", target_os = "linux"))]
86	{
87		// Keep chrome-less on non-macOS - the workbench provides its
88		// own window controls via the WindowsTitleBar / LinuxTitleBar
89		// components. Drag works via `-webkit-app-region: drag` on the
90		// titlebar element.
91		WindowBuilder = WindowBuilder.decorations(false);
92	}
93
94	// Build the main window
95	let MainWindow = WindowBuilder.build().expect("FATAL: Main window build failed");
96
97	// Open DevTools in debug builds
98	#[cfg(debug_assertions)]
99	{
100		MainWindow.open_devtools();
101	}
102
103	MainWindow
104}

106/// Build the initial webview URL, optionally appending `?folder=<path>`
107/// when `~/.land/workspaces/RecentlyOpened.json` has an entry for the
108/// previous session's workspace. Falls back to plain `index.html` if
109/// the file is missing, malformed, or has no resolvable path.
110///
111/// The returned string is already URL-encoded and safe to feed to
112/// `WebviewUrl::External`.
113fn BuildInitialUrl(LocalhostUrl:&str) -> String {
114	let Base = format!("{}/index.html", LocalhostUrl);
115
116	let Recent = match ReadRecentlyOpened() {
117		Ok(Value) => Value,
118		Err(_) => return Base,
119	};
120	let Workspaces = match Recent.get("workspaces").and_then(|V| V.as_array()) {
121		Some(Array) if !Array.is_empty() => Array,
122		_ => return Base,
123	};
124
125	// VS Code's Recently-Opened record can store the folder under a few
126	// different shapes depending on whether the entry came from the
127	// extension host, the workbench, or a `$deltaWorkspaceFolders`
128	// broadcast. Probe them in the same priority order the workbench
129	// itself uses in `getRecentlyOpenedWorkspaces`.
130	let Probe = |Entry:&serde_json::Value| -> Option<String> {
131		// Mountain's own writer emits `{ uri: "file://…", label }` (see
132		// `RecentlyOpened.json` on a freshly closed window). VS Code's
133		// historical `folderUri` / `workspace.configPath` shapes are kept
134		// as fallbacks so imported profiles and third-party writers keep
135		// working.
136		if let Some(Uri) = Entry.get("uri").and_then(|V| V.as_str()) {
137			return Some(Uri.to_string());
138		}
139		if let Some(Uri) = Entry.get("folderUri").and_then(|V| V.as_str()) {
140			return Some(Uri.to_string());
141		}
142		if let Some(Path) = Entry.get("folderUri").and_then(|V| V.get("path")).and_then(|V| V.as_str()) {
143			return Some(Path.to_string());
144		}
145		if let Some(Path) = Entry
146			.get("workspace")
147			.and_then(|V| V.get("configPath"))
148			.and_then(|V| V.get("path"))
149			.and_then(|V| V.as_str())
150		{
151			return Some(Path.to_string());
152		}
153		None
154	};
155
156	let FolderPath = match Workspaces.iter().find_map(Probe) {
157		Some(Path) => Path,
158		None => return Base,
159	};
160
161	// Strip any `file://` scheme so the query param is a plain path
162	// the workbench will stringify into a `file:` URI itself; leaving
163	// the scheme in doubles up and breaks the URL-decode on the other
164	// side (observed as the second `?folder=` boot path appearing as
165	// `file:/Volumes/...` in `wb:boot`).
166	let Normalised = FolderPath.strip_prefix("file://").unwrap_or(FolderPath.as_str()).to_string();
167	if !std::path::Path::new(&Normalised).is_dir() {
168		return Base;
169	}
170
171	let Encoded = url::form_urlencoded::Serializer::new(String::new())
172		.append_pair("folder", &Normalised)
173		.finish();
174	format!("{}/?{}", LocalhostUrl, Encoded)
175}
