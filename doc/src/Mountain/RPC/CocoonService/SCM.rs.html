18pub async fn RegisterScmProvider(
19	Service:&CocoonServiceImpl,
20	req:RegisterScmProviderRequest,
21) -> Result<Response<Empty>, Status> {
22	use CommonLibrary::SourceControlManagement::SourceControlManagementProvider::SourceControlManagementProvider;
23	dev_log!("cocoon", "[CocoonService] Registering SCM provider: {}", req.scm_id);
24
25	// Keep the existing ProviderRegistration bookkeeping so language
26	// feature dispatch can look up the scm provider by handle…
27	let Handle = req
28		.scm_id
29		.as_bytes()
30		.iter()
31		.fold(0u32, |Acc, B| Acc.wrapping_mul(31).wrapping_add(*B as u32));
32	let dto = ProviderRegistrationDTO {
33		Handle,
34		ProviderType:ProviderType::SourceControl,
35		Selector:json!([{ "scmId": req.scm_id }]),
36		SideCarIdentifier:"cocoon-main".to_string(),
37		ExtensionIdentifier:json!(req.extension_id),
38		Options:Some(json!({ "scmId": req.scm_id })),
39	};
40	Service
41		.environment
42		.ApplicationState
43		.Extension
44		.ProviderRegistration
45		.RegisterProvider(Handle, dto);
46
47	// …and additionally route through the
48	// `SourceControlManagementProvider` trait so the SCM state is
49	// materialised in `ApplicationState::SourceControl` (the thing
50	// Sky's SCM view binds to). The prior direct Sky emit bypassed
51	// state tracking, so SCM providers registered by gitlens /
52	// svn etc. never appeared in the SCM view until a
53	// `UpdateScmGroup` call landed - and even then the group-less
54	// header row never materialised.
55	// Pass the same `Handle` used by `ProviderRegistration` above so the
56	// trait keys its marker maps under the same identifier any later
57	// `UpdateScmGroup` lookup will use. See
58	// `SourceControlCreateDTO::Handle` for the wire-shape contract and
59	// `Notification/RegisterScmProvider.rs` for the parallel notification
60	// path that uses Cocoon's sequential handle.
61	let CreateData = json!({
62		"handle": Handle,
63		"id": req.scm_id,
64		"label": req.scm_id,
65		"rootUri": null,
66		"extensionId": req.extension_id,
67	});
68	if let Err(Error) = Service.environment.CreateSourceControl(CreateData).await {
69		dev_log!(
70			"cocoon",
71			"warn: [CocoonService] CreateSourceControl trait failed ({}); falling back to Sky emit",
72			Error
73		);
74		let _ = Service.environment.ApplicationHandle.emit(
75			"sky://scm/register",
76			json!({ "scmId": req.scm_id, "extensionId": req.extension_id }),
77		);
78	}
79
80	Ok(Response::new(Empty {}))
81}

83pub async fn UpdateScmGroup(Service:&CocoonServiceImpl, req:UpdateScmGroupRequest) -> Result<Response<Empty>, Status> {
84	use CommonLibrary::SourceControlManagement::SourceControlManagementProvider::SourceControlManagementProvider;
85	dev_log!(
86		"cocoon",
87		"[CocoonService] update_scm_group: provider={} group={}",
88		req.provider_id,
89		req.group_id
90	);
91
92	let ResourceStates:Vec<serde_json::Value> = req
93		.resource_states
94		.iter()
95		.map(|Rs| {
96			json!({
97				"uri": Rs.uri.as_ref().map(|U| U.value.as_str()).unwrap_or(""),
98				"decorations": Rs.decorations,
99			})
100		})
101		.collect();
102
103	// Re-derive the provider handle from `provider_id` (same hash
104	// RegisterScmProvider used) so the trait can locate the
105	// already-registered provider. `UpdateSourceControlGroup`
106	// mutates the group in state *and* emits the UI event -
107	// downstream Sky components get a deduplicated payload with
108	// group metadata instead of a bare list of URIs.
109	let ProviderHandle = req
110		.provider_id
111		.as_bytes()
112		.iter()
113		.fold(0u32, |Acc, B| Acc.wrapping_mul(31).wrapping_add(*B as u32));
114	let GroupData = json!({
115		"groupId": req.group_id,
116		"label": req.group_id,
117		"resourceStates": ResourceStates,
118	});
119	if let Err(Error) = Service.environment.UpdateSourceControlGroup(ProviderHandle, GroupData).await {
120		dev_log!(
121			"cocoon",
122			"warn: [CocoonService] UpdateSourceControlGroup trait failed ({}); falling back to Sky emit",
123			Error
124		);
125		let _ = Service.environment.ApplicationHandle.emit(
126			"sky://scm/updateGroup",
127			json!({
128				"providerId": req.provider_id,
129				"groupId": req.group_id,
130				"resourceStates": ResourceStates,
131			}),
132		);
133	}
134
135	Ok(Response::new(Empty {}))
136}

138pub async fn GitExec(Service:&CocoonServiceImpl, req:GitExecRequest) -> Result<Response<GitExecResponse>, Status> {
139	dev_log!("cocoon", "[CocoonService] git_exec: {}", req.args.join(" "));
140	dev_log!(
141		"git",
142		"[Git] exec-begin cwd={} args=[{}]",
143		if req.repository_path.is_empty() {
144			"<cwd>".to_string()
145		} else {
146			req.repository_path.clone()
147		},
148		req.args.join(" ")
149	);
150
151	let WorkingDir = if req.repository_path.is_empty() {
152		std::env::current_dir().unwrap_or_default()
153	} else {
154		std::path::PathBuf::from(&req.repository_path)
155	};
156
157	let Output = tokio::process::Command::new("git")
158		.args(&req.args)
159		.current_dir(&WorkingDir)
160		.output()
161		.await
162		.map_err(|Error| {
163			dev_log!("cocoon", "error: [CocoonService] git_exec failed to spawn: {}", Error);
164			dev_log!(
165				"git",
166				"[Git] exec-spawn-fail cwd={:?} args=[{}] error={}",
167				WorkingDir,
168				req.args.join(" "),
169				Error
170			);
171			Status::internal(format!("git_exec: failed to spawn git: {}", Error))
172		})?;
173
174	let ExitCode = Output.status.code().unwrap_or(-1);
175	dev_log!(
176		"cocoon",
177		"[CocoonService] git_exec exit={} stdout={} bytes stderr={} bytes",
178		ExitCode,
179		Output.stdout.len(),
180		Output.stderr.len()
181	);
182	dev_log!(
183		"git",
184		"[Git] exec-done args=[{}] exit={} stdout={} stderr={}",
185		req.args.join(" "),
186		ExitCode,
187		Output.stdout.len(),
188		Output.stderr.len()
189	);
190
191	// Combine stdout lines into repeated string output; prepend stderr lines
192	// with "stderr: " prefix so extension can differentiate them.
193	let StdoutStr = String::from_utf8_lossy(&Output.stdout);
194	let StderrStr = String::from_utf8_lossy(&Output.stderr);
195	let mut OutputLines:Vec<String> = StdoutStr.lines().map(|L| L.to_string()).collect();
196	for Line in StderrStr.lines() {
197		OutputLines.push(format!("stderr: {}", Line));
198	}
199
200	Ok(Response::new(GitExecResponse { output:OutputLines, exit_code:ExitCode }))
201}
